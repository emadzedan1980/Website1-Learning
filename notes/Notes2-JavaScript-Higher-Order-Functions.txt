Higher Order Functions:

a. function that accepts a function as an argument.

	Example: Higher Order Function:

		document.addEventListener('click', myFunction);

		function myFunction(){

			alert('clicked');

		}

b. function the returns a function.

	Example: Non Higher Order Functions:

		function doubleMe(myNumber){

			return myNumber \* 2;

		}

		function tripleMe(myNumber){

			return myNumber \* 3;

		}

		function quadrupleMe(myNumber){

			return myNumber \* 4;

		}

		document.write(doubleMe(5));

		document.write(tripleMe(5));

		document.write(quadrupleMe(5));

	Example: Higher Order Function:

		function createMultiplier(mulitiplier){

			return function(myNumber){

				return myNumber \* mulitiplier

			}

		}

		let  doubleMe = createMultiplier(2);

		let  tripleMe = createMultiplier(3);

		let  quadrupleMe = createMultiplier(4);

		alert(doubleMe(5));

		alert(tripleMe(5));

		alert(quadrupleMe(5));
